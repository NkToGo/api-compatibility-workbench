import com.google.gson.*;
import java.io.*;
import java.nio.charset.StandardCharsets;

/** Container adapter: returns consumer values, never computes assertions. */
public final class PeriodMain {
    public static void main(String[] args) throws Exception {
        var input = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.UTF_8));
        System.out.println("{\"op\":\"read\"}"); System.out.flush();
        var request = JsonParser.parseString(input.readLine()).getAsJsonObject();
        var binding = request.get("binding");
        var value = PeriodConsumer.consume(request.getAsJsonObject("subscription"), binding.isJsonNull() ? null : binding.getAsString());
        var result = new JsonObject(); result.addProperty("op", "return"); result.add("result", value);
        System.out.println(new Gson().toJson(result)); System.out.flush();
        if (input.readLine() == null) throw new IOException("Missing acknowledgement");
    }
}
