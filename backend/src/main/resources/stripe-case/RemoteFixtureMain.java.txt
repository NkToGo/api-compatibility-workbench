final class Rpc {
    private static final Gson JSON = new Gson();
    static synchronized JsonElement call(Object request) {
        try {
            System.out.println(JSON.toJson(request)); System.out.flush();
            var bytes = new ByteArrayOutputStream();
            for (int c; (c = System.in.read()) != '\n';) {
                if (c < 0 || bytes.size() >= 16384) throw new IOException("Protocol closed");
                bytes.write(c);
            }
            return JsonParser.parseString(bytes.toString(StandardCharsets.UTF_8));
        } catch (IOException e) { throw new IllegalStateException("Protocol unavailable"); }
    }
    static List<String> list() {
        var result = new ArrayList<String>();
        for (var item : Rpc.call(Map.of("op", "list")).getAsJsonArray()) result.add(item.getAsString());
        return result;
    }
}
final class RemoteTransport extends HttpClient {
    @Override public StripeResponse request(StripeRequest request) {
        // The SDK also serializes GET params into the URL; the oracle validates the structured params separately.
        String endpoint = request.url().getProtocol() + "://" + request.url().getAuthority() + request.url().getPath();
        var response = Rpc.call(Map.of("op", "provider", "method", request.method().name(), "url", endpoint,
                "version", request.headers().firstValue("Stripe-Version").orElse(""),
                "params", request.params() == null ? Map.of() : request.params()));
        return new StripeResponse(200, HttpHeaders.of(Map.of()), response.toString());
    }
}
